import express from "express";
import Fuse from "fuse.js";

const app = express();
const port = 3000;

// Dữ liệu mẫu
const users = [
    { name: "Minh Thiện" },
    { name: "Thiện Minh" },
    { name: "Quang Hieu" },
    { name: "Anh Tuan" },
    { name: "Tuan Anh" },
    { name: "Hieu Quang" },
];

// Cấu hình Fuse.js
const fuseOptions = {
    keys: ["name"], // Chỉ tìm kiếm trong trường 'name'
    threshold: 0.3, // Độ chính xác tìm kiếm (0.0 khớp hoàn hảo)
};

// Tạo đối tượng Fuse.js
const fuse = new Fuse(users, fuseOptions);

// Middleware để phục vụ các file tĩnh từ thư mục public (chẳng hạn HTML, CSS, JS)
app.use(express.static("public"));

// API tìm kiếm
app.get("/search", (req, res) => {
    const query = req.query.q || ""; // Lấy từ khóa tìm kiếm từ query parameter

    if (query) {
        const results = fuse.search(query).map((result) => result.item); // Tìm kiếm và lấy các kết quả
        res.json(results); // Trả về kết quả tìm kiếm dưới dạng JSON
    } else {
        res.json([]); // Nếu không có query, trả về mảng rỗng
    }
});

// Bắt đầu server
app.listen(port, () => {
    console.log(`Server đang chạy tại http://localhost:${port}`);
});


import React, { useState } from 'react';
import axios from 'axios';

function App() {
  const [searchTerm, setSearchTerm] = useState(''); // Trạng thái lưu từ khóa tìm kiếm
  const [results, setResults] = useState([]); // Trạng thái lưu kết quả tìm kiếm

  // Hàm xử lý sự kiện nhập liệu tìm kiếm
  const handleSearch = async (event) => {
    const query = event.target.value; // Lấy từ khóa tìm kiếm từ input
    setSearchTerm(query); // Cập nhật trạng thái từ khóa tìm kiếm

    if (query) {
      try {
        // Gửi yêu cầu đến API Express
        const response = await axios.get(`http://localhost:3000/search?q=${encodeURIComponent(query)}`);
        setResults(response.data); // Cập nhật kết quả tìm kiếm
      } catch (error) {
        console.error('Lỗi khi gọi API:', error);
      }
    } else {
      setResults([]); // Nếu không có từ khóa, xóa kết quả tìm kiếm
    }
  };

  return (
    <div className="App">
      <h1>Tìm kiếm người dùng</h1>
      {/* Ô input tìm kiếm */}
      <input
        type="text"
        placeholder="Tìm kiếm tên người dùng..."
        value={searchTerm}
        onChange={handleSearch}
      />
      {/* Hiển thị kết quả tìm kiếm */}
      <div id="user-list">
        {results.length > 0 ? (
          results.map((user, index) => (
            <div key={index} className="user-item">
              {user.name}
            </div>
          ))
        ) : (
          <p>Không có kết quả tìm kiếm</p>
        )}
      </div>
    </div>
  );
}

export default App;

Dữ liệu đầu vào:

borderColor: Màu viền của các phần trong biểu đồ.
borderWidth: Độ dày của viền.

responsive: true: Đảm bảo biểu đồ tự động điều chỉnh kích thước khi thay đổi độ phân giải màn hình.
legend: Điều chỉnh vị trí của legend (chú thích biểu đồ).
tooltip: Cài đặt cách hiển thị thông tin khi người dùng hover qua các phần trong biểu đồ.

const express = require('express');
const mysql = require('mysql2');
const multer = require('multer');
const cloudinary = require('./cloudinaryConfig');
const dotenv = require('dotenv');
dotenv.config();

const app = express();
const port = 3000;

// Kết nối với MySQL
const db = mysql.createConnection({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
});

// Kiểm tra kết nối MySQL
db.connect((err) => {
  if (err) {
    console.error('Could not connect to the database:', err);
    return;
  }
  console.log('Connected to the MySQL database');
});

// Cấu hình Multer để lưu trữ file tạm thời
const storage = multer.memoryStorage();
const upload = multer({ storage: storage });

// Route để update hình ảnh
app.post('/update-image/:id', upload.single('image'), (req, res) => {
  const userId = req.params.id;
  const file = req.file;

  if (!file) {
    return res.status(400).send('No file uploaded');
  }

  // Upload file lên Cloudinary
  cloudinary.uploader.upload_stream(
    {
      resource_type: 'auto', // tự động xác định loại file (image, video...)
      folder: 'your-folder', // Nếu bạn muốn sắp xếp hình ảnh vào một thư mục trong Cloudinary
    },
    (error, result) => {
      if (error) {
        return res.status(500).send('Cloudinary upload failed');
      }

      // Lưu đường dẫn hình ảnh vào CSDL
      const imageUrl = result.secure_url;

      const query = 'UPDATE users SET image_url = ? WHERE id = ?';
      db.query(query, [imageUrl, userId], (err, results) => {
        if (err) {
          return res.status(500).send('Error updating database');
        }

        // Trả về kết quả cho client
        res.status(200).send({
          message: 'Image updated successfully',
          imageUrl: imageUrl,
        });
      });
    }
  ).end(file.buffer);
});

// Khởi động server
app.listen(port, () => {
  console.log(`Server running on http://localhost:${port}`);
});
